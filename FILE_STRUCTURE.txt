name: Build Android APK

on:
  push:
    branches: [ main, master ]
  workflow_dispatch:

env:
  GODOT_VERSION: "4.3"

jobs:
  build:
    runs-on: ubuntu-22.04
    steps:
      - name: Checkout
        uses: actions/checkout@v5

      - name: Unpack game project
        run: |
          unzip -oq project.zip
          test -f project.godot
          test -f export_presets.cfg

      - name: Set up JDK 17
        uses: actions/setup-java@v5
        with:
          distribution: temurin
          java-version: "17"

      - name: Set up Android SDK
        uses: android-actions/setup-android@v4

      - name: Install Android SDK packages
        run: |
          yes | sdkmanager --licenses > /dev/null || true
          sdkmanager "platform-tools" "build-tools;34.0.0" "platforms;android-34" "cmdline-tools;latest" "cmake;3.10.2.4988404" "ndk;23.2.8568313"

      - name: Download Godot and export templates
        run: |
          set -e
          BASE="https://github.com/godotengine/godot/releases/download/${GODOT_VERSION}-stable"
          wget -q "$BASE/Godot_v${GODOT_VERSION}-stable_linux.x86_64.zip"
          wget -q "$BASE/Godot_v${GODOT_VERSION}-stable_export_templates.tpz"
          unzip -q "Godot_v${GODOT_VERSION}-stable_linux.x86_64.zip"
          sudo mv "Godot_v${GODOT_VERSION}-stable_linux.x86_64" /usr/local/bin/godot
          sudo chmod +x /usr/local/bin/godot
          TPL_DIR="$HOME/.local/share/godot/export_templates/${GODOT_VERSION}.stable"
          mkdir -p "$TPL_DIR"
          unzip -q "Godot_v${GODOT_VERSION}-stable_export_templates.tpz" -d tpl
          mv tpl/templates/* "$TPL_DIR/"
          ls "$TPL_DIR"

      - name: Create debug keystore
        run: |
          keytool -genkeypair -v -keystore "$HOME/debug.keystore" -storepass android -alias androiddebugkey -keypass android -keyalg RSA -keysize 2048 -validity 10000 -dname "CN=Android Debug,O=Android,C=US"

      - name: Configure Godot editor settings
        run: |
          mkdir -p "$HOME/.config/godot"
          cat > "$HOME/.config/godot/editor_settings-4.3.tres" <<EOT
          [gd_resource type="EditorSettings" format=3]

          [resource]
          export/android/android_sdk_path = "$ANDROID_SDK_ROOT"
          export/android/java_sdk_path = "$JAVA_HOME"
          export/android/debug_keystore = "$HOME/debug.keystore"
          export/android/debug_keystore_user = "androiddebugkey"
          export/android/debug_keystore_pass = "android"
          export/android/force_system_user = false
          export/android/shutdown_adb_on_exit = true
          EOT
          cat "$HOME/.config/godot/editor_settings-4.3.tres"

      - name: Import project
        run: godot --headless --path . --import 2>&1 | tee import.log || true

      - name: Script check (prints GDScript errors)
        continue-on-error: true
        run: godot --headless --path . --quit-after 120 2>&1 | tee run.log

      - name: Export APK
        run: |
          set -o pipefail
          mkdir -p build
          godot --headless --path . --export-debug "Android" build/HighwayDrive.apk 2>&1 | tee export.log
          ls -la build

      - name: Upload logs
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: build-logs
          path: |
            import.log
            run.log
            export.log
          if-no-files-found: ignore

      - name: Upload APK artifact
        uses: actions/uploa
